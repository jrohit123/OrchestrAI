# ── OTP GENERATION ───────────────────────────────────
import random, hashlib, datetime
from db import execute

def generate_otp(user_id: str, action_context: dict) -> str:
    # 4-digit random PIN
    otp = str(random.randint(1000, 9999))

    # Never store raw OTP — store SHA-256 hash only
    otp_hash = hashlib.sha256(otp.encode()).hexdigest()

    expiry = datetime.datetime.utcnow() + datetime.timedelta(minutes=3)

    # Invalidate any previous unused OTPs for this user
    execute("""
        UPDATE otp_tokens SET used = true
        WHERE user_id = %s AND used = false
    """, [user_id])

    # Insert new OTP record
    execute("""
        INSERT INTO otp_tokens
        (user_id, otp_hash, action_context, expires_at, used)
        VALUES (%s, %s, %s, %s, false)
    """, [user_id, otp_hash, json.dumps(action_context), expiry])

    return otp  # Only time raw OTP exists — passed to email sender

import requests

        "htmlContent": build_otp_email_html(otp, action_desc, user_name, org_name)

# ── OTP VERIFICATION ─────────────────────────────────
def verify_otp(user_id: str, entered_otp: str) -> dict:
    entered_hash = hashlib.sha256(entered_otp.encode()).hexdigest()
    now = datetime.datetime.utcnow()

    row = execute("""
        SELECT id, action_context, expires_at
        FROM otp_tokens
        WHERE user_id = %s
          AND otp_hash = %s
          AND used = false
          AND expires_at > %s
        LIMIT 1
    """, [user_id, entered_hash, now])

    if not row:
        return {"valid": False,
                "reason": "Invalid code, already used, or expired"}

    # Mark used IMMEDIATELY — single use enforced
    execute("UPDATE otp_tokens SET used = true WHERE id = %s", [row["id"]])

    return {"valid": True, "action_context": row["action_context"]}

Security Story for the Client

How to frame this for the jewellery client. The honest comparison: their current process has weaker security than what you're proposing.

❌ Today — Without OrchestrAI
🔓Shared Tally logins — multiple people using same credentials
🔓Invoice created by anyone who knows the password
🔓Zero audit trail — no log of who created what, when
🔓WhatsApp groups used as informal approvals — no verification
🔓New employee gets full system access from Day 1
🔓No anomaly detection — fraud discovered after the fact
✅ With OrchestrAI
🔐Every user individually identified by verified phone number
🔐High-value invoices require email OTP — even compromised WhatsApp can't proceed without email access
🔐Immutable audit log — every action, user, timestamp, outcome
🔐Formal approval gates — MD approves via WhatsApp button, logged permanently
🔐Role-based access — warehouse staff cannot create invoices. Period.
🔐Anomaly alerts — unusual activity flagged to admin instantly
📧
Out-of-band OTP
Code sent to email — separate channel from WhatsApp. Compromised WhatsApp alone is not enough to authorize a transaction.
Brevo → registered email → user types in WhatsApp
⏱️
3-Minute Expiry
OTP automatically invalidated after 3 minutes. Useless to anyone who finds it in email history later.
expires_at checked on every verify attempt
1️⃣
Single-Use Enforcement
Marked used=true the instant it's verified successfully. Same code can never be replayed even within the 3-minute window.
DB update before workflow resumes
🔒
Hash-Only Storage
Raw OTP never stored in the database. Only SHA-256 hash stored. Even a DB breach reveals nothing usable.
hashlib.sha256 · raw OTP discarded after email send
🧹
Previous OTPs Invalidated
When a new OTP is generated, all previous unused OTPs for that user are immediately invalidated. No stale codes floating around.
UPDATE otp_tokens SET used=true on new generation
⚙️
Admin-Configurable
Admin decides which flows and which thresholds require OTP. Not a blanket gate — applied intelligently where risk is highest.
otp_required field in workflows table

-- New table: OTP tokens
CREATE TABLE otp_tokens (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id         UUID NOT NULL REFERENCES users(id),
    otp_hash        TEXT NOT NULL,
    action_context  JSONB,
    expires_at      TIMESTAMPTZ NOT NULL,
    used            BOOLEAN DEFAULT FALSE,
    attempts        INTEGER DEFAULT 0,
    created_at      TIMESTAMPTZ DEFAULT NOW()
);
CREATE INDEX idx_otp_user_expiry ON otp_tokens(user_id, expires_at)
    WHERE used = false;

-- Modified: workflows — add OTP config fields
ALTER TABLE workflows
    ADD COLUMN otp_required   BOOLEAN DEFAULT false,
    ADD COLUMN otp_threshold  DECIMAL(12,2) DEFAULT NULL;

-- Modified: users — enforce email NOT NULL
ALTER TABLE users
    ALTER COLUMN email SET NOT NULL;

-- Modified: audit_log — track OTP usage
ALTER TABLE audit_log
    ADD COLUMN otp_used BOOLEAN DEFAULT false;

-- Seed: enable OTP for create_invoice above ₹50,000
UPDATE workflows
SET otp_required = true, otp_threshold = 50000.00
WHERE intent_key = 'create_invoice';
